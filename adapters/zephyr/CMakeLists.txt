if(CONFIG_FAULT_INJECTION)

  # Real enforcement of "must not ship in a non-test build": a Kconfig
  # bool default cannot fail a build by itself, so this is a CMake-time
  # hard error instead. CONFIG_ZTEST is what Zephyr sets for any Ztest
  # test image, which is the only intended consumer of this module today.
  # Carried forward unchanged from C-MSP's CMakeLists.txt.
  if(NOT CONFIG_ZTEST)
    message(FATAL_ERROR
      "CONFIG_FAULT_INJECTION=y outside a Ztest build. This module's "
      "design requires fault injection to be physically absent from "
      "non-test images -- see docs/planning.md and Kconfig for why. If "
      "you have a legitimate new consumer, that's a deliberate decision "
      "to make, not something to bypass here.")
  endif()

  # ZEPHYR_FAULT_INJECT_MODULE_DIR is generated by west from this
  # module's `name: fault_inject` in ../../zephyr/module.yml -- it points
  # at the module's top-level directory (the repo root), regardless of
  # where this CMakeLists.txt itself lives. That's what lets the
  # portable core's include/ and src/ stay adapter-agnostic: this is the
  # only CMakeLists.txt that knows the core lives one level up.
  zephyr_include_directories(${ZEPHYR_FAULT_INJECT_MODULE_DIR}/include)
  zephyr_library()
  zephyr_library_sources(
    ${ZEPHYR_FAULT_INJECT_MODULE_DIR}/src/fault_inject.c
    ${CMAKE_CURRENT_LIST_DIR}/fi_port_zephyr.c
  )

endif()
